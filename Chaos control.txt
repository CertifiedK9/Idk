                                                                                                                                                                                                                                          const mongoose = require("mongoose");
                                                                                                                                                                                                                                          const http = require("http");
                                                                                                                                                                                                                                          const { Server } = require("socket.io");

                                                                                                                                                                                                                                          const app = express();
                                                                                                                                                                                                                                          const server = http.createServer(app);
                                                                                                                                                                                                                                          const io = new Server(server);

                                                                                                                                                                                                                                          app.use(express.json());
                                                                                                                                                                                                                                          app.use(express.static("public"));

                                                                                                                                                                                                                                          // Connect to local MongoDB
                                                                                                                                                                                                                                          mongoose.connect("mongodb://127.0.0.1:27017/clanSystem");

                                                                                                                                                                                                                                          // ==================== MODELS ====================

                                                                                                                                                                                                                                          // Users for ranking
                                                                                                                                                                                                                                          const User = mongoose.model("User", {
                                                                                                                                                                                                                                            username: String,
                                                                                                                                                                                                                                              points: { type: Number, default: 0 },
                                                                                                                                                                                                                                                clan: String
                                                                                                                                                                                                                                                });

                                                                                                                                                                                                                                                // Auctions
                                                                                                                                                                                                                                                const Auction = mongoose.model("Auction", {
                                                                                                                                                                                                                                                  item: String,
                                                                                                                                                                                                                                                    highestBid: { type: Number, default: 0 },
                                                                                                                                                                                                                                                      highestBidder: String,
                                                                                                                                                                                                                                                        endTime: Date
                                                                                                                                                                                                                                                        });

                                                                                                                                                                                                                                                        // Clans
                                                                                                                                                                                                                                                        const Clan = mongoose.model("Clan", {
                                                                                                                                                                                                                                                          name: String,
                                                                                                                                                                                                                                                            members: [String]
                                                                                                                                                                                                                                                            });

                                                                                                                                                                                                                                                            // ==================== ROUTES ====================

                                                                                                                                                                                                                                                            // Register user
                                                                                                                                                                                                                                                            app.post("/register", async (req, res) => {
                                                                                                                                                                                                                                                              const user = await User.create({ username: req.body.username });
                                                                                                                                                                                                                                                                res.json(user);
                                                                                                                                                                                                                                                                });

                                                                                                                                                                                                                                                                // Leaderboard
                                                                                                                                                                                                                                                                app.get("/leaderboard", async (req, res) => {
                                                                                                                                                                                                                                                                  const users = await User.find().sort({ points: -1 });
                                                                                                                                                                                                                                                                    res.json(users);
                                                                                                                                                                                                                                                                    });

                                                                                                                                                                                                                                                                    // Create clan
                                                                                                                                                                                                                                                                    app.post("/clan", async (req, res) => {
                                                                                                                                                                                                                                                                      const clan = await Clan.create({ name: req.body.name, members: [] });
                                                                                                                                                                                                                                                                        res.json(clan);
                                                                                                                                                                                                                                                                        });

                                                                                                                                                                                                                                                                        // Join clan
                                                                                                                                                                                                                                                                        app.post("/clan/join", async (req, res) => {
                                                                                                                                                                                                                                                                          const clan = await Clan.findOne({ name: req.body.clan });
                                                                                                                                                                                                                                                                            clan.members.push(req.body.username);
                                                                                                                                                                                                                                                                              await clan.save();
                                                                                                                                                                                                                                                                                res.json(clan);
                                                                                                                                                                                                                                                                                });

                                                                                                                                                                                                                                                                                // Create auction
                                                                                                                                                                                                                                                                                app.post("/auction", async (req, res) => {
                                                                                                                                                                                                                                                                                  const auction = await Auction.create({
                                                                                                                                                                                                                                                                                      item: req.body.item,
                                                                                                                                                                                                                                                                                          endTime: new Date(Date.now() + 60000)
                                                                                                                                                                                                                                                                                            });
                                                                                                                                                                                                                                                                                              res.json(auction);
                                                                                                                                                                                                                                                                                              });

                                                                                                                                                                                                                                                                                              // Place bid
                                                                                                                                                                                                                                                                                              app.post("/bid", async (req, res) => {
                                                                                                                                                                                                                                                                                                const auction = await Auction.findById(req.body.id);
                                                                                                                                                                                                                                                                                                  if (req.body.amount > auction.highestBid) {
                                                                                                                                                                                                                                                                                                      auction.highestBid = req.body.amount;
                                                                                                                                                                                                                                                                                                          auction.highestBidder = req.body.username;
                                                                                                                                                                                                                                                                                                              await auction.save();
                                                                                                                                                                                                                                                                                                                }
                                                                                                                                                                                                                                                                                                                  res.json(auction);
                                                                                                                                                                                                                                                                                                                  });

                                                                                                                                                                                                                                                                                                                  // ==================== LIVE CHAT ====================

                                                                                                                                                                                                                                                                                                                  io.on("connection", (socket) => {
                                                                                                                                                                                                                                                                                                                    socket.on("chatMessage", (msg) => {
                                                                                                                                                                                                                                                                                                                        io.emit("chatMessage", msg);
                                                                                                                                                                                                                                                                                                                          });
                                                                                                                                                                                                                                                                                                                          });

                                                                                                                                                                                                                                                                                                                          // ==================== START SERVER ====================

                                                                                                                                                                                                                                                                                                                          server.listen(3000, () => {
                                                                                                                                                                                                                                                                                                                            console.log("Server running on http://localhost:3000");
